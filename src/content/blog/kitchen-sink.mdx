## Code

Expressive Code adds titles, line highlighting and diffs:

```c title="gpio_read.c" {12-13}
#include <gpiod.h>
#include <stdio.h>

int main(void)
{
	unsigned int offset = 17;
	struct gpiod_chip *chip = gpiod_chip_open("/dev/gpiochip0");
	struct gpiod_line_settings *settings = gpiod_line_settings_new();
	struct gpiod_line_config *line_cfg = gpiod_line_config_new();

	gpiod_line_settings_set_direction(settings, GPIOD_LINE_DIRECTION_INPUT);
	gpiod_line_config_add_line_settings(line_cfg, &offset, 1, settings);
	struct gpiod_line_request *req = gpiod_chip_request_lines(chip, NULL, line_cfg);

	printf("GPIO %u = %d\n", offset, gpiod_line_request_get_value(req, offset));
	return 0;
}
```
